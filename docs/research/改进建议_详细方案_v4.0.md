### P0-1 修复: 消除 time.sleep() 阻塞

#### 问题回顾

#### 改进方案

**方案A: 使用 asyncio.sleep()（推荐）**

将 `game_controller.py` 中的游戏循环改为异步：

```python

class OriginValidator:
    def __init__(self, allowed_origins: list[str] | None = None):
        if allowed_origins is None:
            # 默认拒绝所有，强制显式配置
            raise ValueError(
                "allowed_origins must be explicitly configured. "
                "Use ['localhost'] for development."
            )
        if "*" in allowed_origins:
            import warnings
            warnings.warn(
                "Wildcard origin '*' is insecure. "
                "Use specific origins in production.",
                SecurityWarning,
                stacklevel=2
            )
        self.allowed_origins = allowed_origins

    def validate(self, origin: str | None) -> bool:
        if origin is None:
            return False  # 拒绝无Origin的连接
        if "*" in self.allowed_origins:
            return True
        return origin in self.allowed_origins
```

### P1-2 修复: 实现服务器游戏逻辑

#### 改进方案

```python
class GameRoom:
    """一个游戏房间 = 一局游戏"""

    def __init__(self, room_id: str, config: GameConfig):
        self.room_id = room_id
        self.engine = create_standard_engine(config)
        self.players: dict[str, WebSocket] = {}
        self.state = RoomState.WAITING

    async def handle_player_action(self, player_id: str, action: dict):
        """处理玩家动作 — 权威服务器验证"""
        # 1. 验证是否轮到该玩家
        if not self.engine.turns.is_current_player(player_id):
            await self.send_error(player_id, "不是你的回合")
            return

        # 2. 验证动作合法性
        if not self.engine.validate_action(player_id, action):
            await self.send_error(player_id, "非法操作")
            return

        # 3. 执行动作
        result = await self.engine.execute_action(player_id, action)

        # 4. 广播状态更新到所有玩家
        for pid, ws in self.players.items():
            state = self.engine.get_player_view(pid)  # 各玩家看到不同视图
            await ws.send(json.dumps({"type": "state_update", "state": state}))

    async def handle_disconnect(self, player_id: str):
        """断线处理"""
        self.state_snapshots[player_id] = self.engine.snapshot()
        # AI接管
        self.engine.set_ai_control(player_id)
        await self.broadcast({"type": "player_disconnected", "player": player_id})

    async def handle_reconnect(self, player_id: str, ws: WebSocket):
        """重连处理"""
        self.players[player_id] = ws
        self.engine.remove_ai_control(player_id)
        # 发送完整状态
        state = self.engine.get_player_view(player_id)
        await ws.send(json.dumps({"type": "full_state", "state": state}))
```

### P1-3 修复: EventBus支持异步handler和队列化

#### 改进方案

```python
import asyncio
from collections import deque

class AsyncEventBus:
    """异步事件总线 — 队列化处理防止顺序问题"""

    def __init__(self):
        self._handlers: dict[str, list[tuple[int, Callable]]] = {}
        self._queue: deque = deque()
        self._processing = False

    def on(self, event: str, handler: Callable, priority: int = 0):
        """注册事件处理器"""
        if event not in self._handlers:
            self._handlers[event] = []
        self._handlers[event].append((priority, handler))
        self._handlers[event].sort(key=lambda x: x[0], reverse=True)

    async def emit(self, event: str, *args, **kwargs):
        """发射事件 — 队列化处理"""
        self._queue.append((event, args, kwargs))
        if not self._processing:
            await self._process_queue()

    async def _process_queue(self):
        """按队列顺序处理事件"""
        self._processing = True
        try:
            while self._queue:
                event, args, kwargs = self._queue.popleft()
                handlers = self._handlers.get(event, [])
                for priority, handler in handlers:
                    if asyncio.iscoroutinefunction(handler):
                        await handler(*args, **kwargs)
                    else:
                        handler(*args, **kwargs)
        finally:
            self._processing = False
```

### P1-9 修复: 引入回合阶段状态机

#### 改进方案

使用 `python-statemachine` 或自定义Enum状态机：

```python
from enum import Enum, auto

class TurnPhase(Enum):
    PREPARE = auto()      # 准备阶段
    JUDGE = auto()        # 判定阶段
    DRAW = auto()         # 摸牌阶段
    PLAY = auto()         # 出牌阶段
    DISCARD = auto()      # 弃牌阶段
    FINISH = auto()       # 结束阶段

class TurnStateMachine:
    """回合阶段状态机"""

    TRANSITIONS = {
        TurnPhase.PREPARE: TurnPhase.JUDGE,
        TurnPhase.JUDGE: TurnPhase.DRAW,
        TurnPhase.DRAW: TurnPhase.PLAY,
        TurnPhase.PLAY: TurnPhase.DISCARD,
        TurnPhase.DISCARD: TurnPhase.FINISH,
        TurnPhase.FINISH: None,  # 回合结束
    }

    def __init__(self, event_bus: AsyncEventBus):
        self.phase = TurnPhase.PREPARE
        self.event_bus = event_bus
        self.skip_phases: set[TurnPhase] = set()

    async def advance(self):
        """推进到下一阶段"""
        next_phase = self.TRANSITIONS[self.phase]
        if next_phase is None:
            return False  # 回合结束

        # 发射离开当前阶段事件
        await self.event_bus.emit(f"phase_exit_{self.phase.name}", self.phase)

        # 检查是否跳过（如乐不思蜀跳过PLAY）
        if next_phase in self.skip_phases:
            self.skip_phases.discard(next_phase)
            self.phase = next_phase
            return await self.advance()

        self.phase = next_phase
        # 发射进入新阶段事件
        await self.event_bus.emit(f"phase_enter_{self.phase.name}", self.phase)
        return True

    def skip_next(self, phase: TurnPhase):
        """标记跳过某阶段（如乐不思蜀）"""
        self.skip_phases.add(phase)
```

### P2-8 修复: 规则优先级系统

```python
class RulePriority(Enum):
    BASIC_RULE = 0       # 基本游戏规则
    CARD_EFFECT = 10     # 游戏牌效果
    HERO_SKILL = 20      # 武将技能
    LORD_SKILL = 30      # 主公技

class RuleResolver:
    """规则优先级解析器"""

    def resolve_conflict(self, rules: list[Rule]) -> Rule:
        """当多条规则冲突时，取最高优先级"""
        return max(rules, key=lambda r: r.priority.value)

    def can_play_sha(self, player) -> bool:
        """是否可以出杀 — 示例"""
        base_rule = Rule(RulePriority.BASIC_RULE, max_sha=1)

        # 检查武将技能是否覆盖
        for skill in player.skills:
            if skill.overrides("max_sha"):
                skill_rule = Rule(RulePriority.HERO_SKILL, max_sha=skill.max_sha)
                base_rule = self.resolve_conflict([base_rule, skill_rule])

        # 检查装备效果（诸葛连弩）
        if player.weapon and player.weapon.overrides("max_sha"):
            equip_rule = Rule(RulePriority.CARD_EFFECT, max_sha=float('inf'))
            base_rule = self.resolve_conflict([base_rule, equip_rule])

        return player.sha_used_this_turn < base_rule.max_sha
```
